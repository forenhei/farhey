#include <iostream>
#include <string>
#include <vector>
#include <cstdlib>
#include <ctime>

using namespace std;

int main() {
    setlocale(LC_ALL, "");

    srand(time(0));

    vector<string> answers = {
        "Я слышу тебя. Расскажи, что тебя волнует.",
        "Иногда ответ находится не снаружи, а внутри тебя.",
        "Интересный вопрос. Подумай, какой ответ ты сам считаешь правильным.",
        "Не бойся задавать вопросы. Именно с них начинается поиск.",
        "У каждого выбора есть последствия. Выбирай осознанно.",
        "Даже маленький шаг может изменить многое.",
        "Я не могу прожить жизнь за тебя, но могу выслушать.",
        "Иногда нужно просто остановиться и подумать.",
        "Будущее ещё не написано.",
        "Что для тебя сейчас действительно важно?"
    };

    string message;

    cout << "========================================\n";
    cout << "        THE GOD - CHAT SYSTEM\n";
    cout << "========================================\n";
    cout << "Ты вошёл в чат с вымышленным Богом.\n";
    cout << "Напиши 'выход', чтобы завершить разговор.\n";
    cout << "----------------------------------------\n\n";

    cout << "Бог: Я здесь. Что ты хочешь спросить?\n\n";

    while (true) {
        cout << "Ты: ";
        getline(cin, message);

        if (message == "выход" || message == "exit") {
            cout << "\nБог: До следующего разговора.\n";
            break;
        }

        if (message.empty()) {
            cout << "Бог: Ты ничего не сказал.\n\n";
            continue;
        }

        int randomAnswer = rand() % answers.size();

        cout << "Бог: " << answers[randomAnswer] << "\n\n";
    }

    return 0;
}
